// Render the sync page
function gsf_sync_members_page() {
    if (!current_user_can('manage_options')) {
        wp_die(__('You do not have sufficient permissions to access this page.'));
    }

    // Handle sync settings update
    if (isset($_POST['gsf_save_sync_settings']) && check_admin_referer('gsf_sync_settings_nonce')) {
        $sync_interval = intval($_POST['gsf_sync_interval']);
        $enable_auto_sync = isset($_POST['gsf_enable_auto_sync']) ? 1 : 0;
        
        // Validate sync interval (minimum 15 minutes, maximum 24 hours)
        if ($sync_interval < 900) $sync_interval = 900; // 15 minutes
        if ($sync_interval > 86400) $sync_interval = 86400; // 24 hours
        
        // Save settings
        update_option('gsf_sync_interval', $sync_interval);
        update_option('gsf_enable_auto_sync', $enable_auto_sync);
        
        // Update the cron job
        gsf_schedule_sync_cron();
        
        add_settings_error(
            'gsf_messages',
            'gsf_message',
            __('Sync settings saved successfully.'),
            'updated'
        );
    }

    // Handle force reschedule (debug feature)
    if (isset($_POST['gsf_force_reschedule']) && check_admin_referer('gsf_force_reschedule_nonce') && gsf_user_has_platform_email()) {
        wp_clear_scheduled_hook('gsf_sync_members_cron');
        gsf_schedule_sync_cron();
        
        add_settings_error(
            'gsf_messages',
            'gsf_message',
            __('Cron job forcefully rescheduled.'),
            'updated'
        );
    }

    // Handle manual sync
    if (isset($_POST['gsf_sync_now']) && check_admin_referer('gsf_sync_members_nonce')) {
        try {
            $api = new ZohoAPI();
            $result = $api->getMembers(1, 200, [], true); // Force sync for manual requests
            
            // Store sync results in a transient (valid for 24 hours)
            set_transient('gsf_last_sync_results', [
                'timestamp' => time(),
                'success' => true,
                'error' => '',
                'created' => isset($result['sync_stats']['created']) ? $result['sync_stats']['created'] : 0,
                'updated' => isset($result['sync_stats']['updated']) ? $result['sync_stats']['updated'] : 0,
                'deleted' => isset($result['deleted_count']) ? $result['deleted_count'] : 0,
                'failed' => isset($result['sync_stats']['failed']) ? $result['sync_stats']['failed'] : 0,
                'total_fetched' => isset($result['sync_stats']['total_fetched']) ? $result['sync_stats']['total_fetched'] : 0,
            ], DAY_IN_SECONDS);
            
            // Redirect to the detailed results page
            wp_safe_redirect(admin_url('admin.php?page=gsf-sync-results'));
            exit;
            
        } catch (Exception $e) {
            // Store error information
            set_transient('gsf_last_sync_results', [
                'timestamp' => time(),
                'success' => false,
                'error' => $e->getMessage(),
                'created' => 0,
                'updated' => 0,
                'deleted' => 0,
                'failed' => 0
            ], DAY_IN_SECONDS);
            
            add_settings_error(
                'gsf_messages',
                'gsf_message',
                __('Error syncing with Zoho: ') . $e->getMessage(),
                'error'
            );
        }
    }

    // Display status messages
    settings_errors('gsf_messages');

    // Get last sync time
    $last_sync = get_option('gsf_zoho_last_sync');
    $last_sync_display = $last_sync ? wp_date('F j, Y \a\t g:i a', $last_sync) : 'Never';
    
    // Get current settings and ensure cron is scheduled if needed
    $sync_interval = get_option('gsf_sync_interval', 3600); // Default 1 hour
    $enable_auto_sync = get_option('gsf_enable_auto_sync', 1); // Default enabled
    $next_scheduled = wp_next_scheduled('gsf_sync_members_cron');
    
    // Auto-schedule if missing and auto-sync is enabled
    if ($enable_auto_sync && !$next_scheduled) {
        gsf_schedule_sync_cron();
        $next_scheduled = wp_next_scheduled('gsf_sync_members_cron');
    }
    ?>
    <div class="wrap">
        <h1><?php echo esc_html(get_admin_page_title()); ?></h1>
        <p><?php printf(__('Last synchronized with Zoho: %s'), '<strong>' . esc_html($last_sync_display) . '</strong>'); ?></p>
        
        <div class="postbox" style="margin-top: 20px;">
            <div class="postbox-header" style="padding: 10px; cursor: default;">
                <h2 class="hndle" ><?php _e('Manual Sync'); ?></h2>
            </div>
            <div class="inside">
                <form method="post" action="">
                    <?php wp_nonce_field('gsf_sync_members_nonce'); ?>
                    <p><?php _e('Click the button below to manually sync members with Zoho CRM.'); ?></p>
                    <p>
                        <input type="submit" name="gsf_sync_now" class="button button-primary" value="<?php esc_attr_e('Sync with Zoho Now'); ?>">
                        
                        <?php if (gsf_user_has_platform_email()): ?>
                        <a href="<?php echo admin_url('admin-ajax.php?action=gsf_trigger_cron'); ?>" 
                           class="button button-secondary" 
                           style="margin-left: 10px;"
                           onclick="return confirm('This will trigger the automatic cron sync (respects intervals). Continue?');">
                            <?php _e('Trigger Cron Job'); ?>
                        </a>
                        <a href="<?php echo admin_url('admin-ajax.php?action=gsf_trigger_cron&force=1'); ?>" 
                           class="button button-secondary" 
                           style="margin-left: 5px;"
                           onclick="return confirm('This will force trigger the cron sync (bypasses intervals). Continue?');">
                            <?php _e('Force Cron Sync'); ?>
                        </a>
                        <?php endif; ?>
                    </p>
                </form>
                
                <?php if (gsf_user_has_platform_email()): ?>
                <div class="notice notice-info inline" style="margin: 15px 0 0 0;">
                    <p><strong>Development Cron Options:</strong></p>
                    <ul style="margin-left: 20px;">
                        <li><strong>Manual Trigger:</strong> Use the "Trigger Cron Job" button above</li>
                        <li><strong>URL Trigger:</strong> Visit <code><?php echo admin_url('admin-ajax.php?action=gsf_trigger_cron'); ?></code></li>
                        <li><strong>WordPress Cron:</strong> Visit <code><?php echo site_url('wp-cron.php'); ?></code></li>
                        <li><strong>System Cron:</strong> Add to your system cron: <code>*/15 * * * * curl -s <?php echo site_url('wp-cron.php'); ?> > /dev/null 2>&1</code></li>
                    </ul>
                </div>
                <?php endif; ?>
            </div>

            <div class="postbox-header" style="padding: 10px; cursor: default;">
                <h2 class="hndle"><?php _e('Automatic Sync Settings'); ?></h2>
            </div>
            <div class="inside">
                <form method="post" action="">
                    <?php wp_nonce_field('gsf_sync_settings_nonce'); ?>
                    <table class="form-table" role="presentation">
                        <tbody>
                            <tr>
                                <th scope="row">
                                    <label for="gsf_enable_auto_sync"><?php _e('Enable Automatic Sync'); ?></label>
                                </th>
                                <td>
                                    <label for="gsf_enable_auto_sync">
                                        <input type="checkbox" name="gsf_enable_auto_sync" id="gsf_enable_auto_sync" value="1" <?php checked($enable_auto_sync, 1); ?> />

/**
 * Add custom cron interval - must be hooked early
 */
function gsf_add_custom_cron_interval($schedules) {
    $sync_interval = get_option('gsf_sync_interval', 3600);
    
    $schedules['gsf_sync_interval_custom'] = array(
        'interval' => $sync_interval,
        'display' => sprintf(__('GSF Sync Every %s'), human_time_diff(0, $sync_interval))
    );
    
    return $schedules;
}
// Hook the cron intervals filter early
add_filter('cron_schedules', 'gsf_add_custom_cron_interval');

/**
 * Schedule the sync cron job based on current settings
 */
function gsf_schedule_sync_cron() {
    // Clear any existing scheduled event
    wp_clear_scheduled_hook('gsf_sync_members_cron');
    
    $enable_auto_sync = get_option('gsf_enable_auto_sync', 1);
    
    if ($enable_auto_sync) {
        // Schedule new event using our custom interval
        $scheduled = wp_schedule_event(time(), 'gsf_sync_interval_custom', 'gsf_sync_members_cron');
        
        if ($scheduled === false) {
            $logger = GSF_Logger::getInstance();
            $logger->log('Failed to schedule sync cron job', 'ERROR');
        } else {
            $logger = GSF_Logger::getInstance();
            $logger->log('Sync cron job scheduled successfully', 'INFO', [
                'next_run' => wp_next_scheduled('gsf_sync_members_cron'),
                'interval' => get_option('gsf_sync_interval', 3600)
            ]);
        }
    }
}

/**
 * Handle the cron job execution
 */
function gsf_handle_sync_cron() {
    $logger = GSF_Logger::getInstance();
    $logger->log('Cron job triggered', 'INFO', [
        'time' => wp_date('Y-m-d H:i:s'),
        'timestamp' => time()
    ]);
    
    $enable_auto_sync = get_option('gsf_enable_auto_sync', 1);
    
    if (!$enable_auto_sync) {
        $logger->log('Cron job aborted - auto sync disabled', 'INFO');
        return;
    }
    
    try {
        $api = new ZohoAPI();
        
        // For cron jobs, we want to respect the configured sync interval but not the manual sync rate limiting
        // Check if enough time has passed based on the configured interval
        $last_sync = get_option('gsf_zoho_last_sync', 0);
        $sync_interval = get_option('gsf_sync_interval', 3600);
        $time_since_last = time() - $last_sync;
        
        if ($time_since_last >= $sync_interval) {
            $logger->log('Cron sync proceeding - interval met', 'INFO', [
                'time_since_last_sync' => $time_since_last,
                'required_interval' => $sync_interval
            ]);
            
            $result = $api->getMembers(1, 200, [], true); // Force sync for cron jobs
            
            // Log the cron sync results
            $logger->log('Automatic sync completed via cron', 'INFO', [
                'created' => isset($result['sync_stats']['created']) ? $result['sync_stats']['created'] : 0,
                'updated' => isset($result['sync_stats']['updated']) ? $result['sync_stats']['updated'] : 0,
                'deleted' => isset($result['deleted_count']) ? $result['deleted_count'] : 0,
                'failed' => isset($result['sync_stats']['failed']) ? $result['sync_stats']['failed'] : 0,
                'last_sync_before' => $last_sync,
                'last_sync_after' => get_option('gsf_zoho_last_sync', 0)
            ]);
        } else {
            $wait_time = $sync_interval - $time_since_last;
            $logger->log('Cron sync skipped - interval not met', 'INFO', [
                'time_since_last_sync' => $time_since_last,
                'required_interval' => $sync_interval,
                'wait_seconds_remaining' => $wait_time
            ]);
        }
        
    } catch (Exception $e) {
        $logger->log('Automatic sync failed via cron', 'ERROR', [
            'error' => $e->getMessage(),
            'trace' => $e->getTraceAsString()
        ]);
    }
}

// Hook up the cron action
add_action('gsf_sync_members_cron', 'gsf_handle_sync_cron');

// Schedule cron on theme activation/init
add_action('after_switch_theme', 'gsf_schedule_sync_cron');
add_action('init', function() {
    // Ensure cron is scheduled if it's not already
    if (get_option('gsf_enable_auto_sync', 1) && !wp_next_scheduled('gsf_sync_members_cron')) {
        gsf_schedule_sync_cron();
    }
});
