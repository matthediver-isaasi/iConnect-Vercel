create table public.booking (
  id uuid not null default gen_random_uuid (),
  event_id uuid not null,
  member_id uuid null,
  attendee_email text null,
  attendee_first_name text null,
  attendee_last_name text null,
  ticket_price numeric not null,
  payment_method text null,
  booking_reference text null,
  status text null default 'confirmed'::text,
  confirmation_token text null,
  backstage_order_id text null,
  zoom_registrant_id text null,
  po_to_follow boolean not null default false,
  is_one_off_event boolean null default false,
  voucher_amount numeric null default 0,
  training_fund_amount numeric null default 0,
  account_amount numeric null default 0,
  stripe_payment_intent_id text null,
  total_cost numeric null default 0,
  organization_id uuid null,
  purchase_order_number text null,
  ticket_class_id character varying(255) null,
  ticket_class_name character varying(255) null,
  booking_group_reference text null,
  xero_invoice_id text null,
  xero_invoice_number text null,
  xero_invoice_pdf_base64 text null,
  is_guest_booking boolean not null default false,
  created_at timestamp with time zone null,
  tenant_id uuid null,
  guest_organisation_name text null,
  attendee_phone text null,
  attendee_job_title text null,
  discount_code_id uuid null,
  discount_code_amount numeric null default 0,
  xero_credit_note_id text null,
  xero_credit_note_number text null,
  third_party_consent boolean null,
  constraint booking_pkey primary key (id),
  constraint booking_confirmation_token_unique unique (confirmation_token),
  constraint booking_booking_reference_unique unique (booking_reference),
  constraint booking_tenant_id_fkey foreign KEY (tenant_id) references tenant (id),
  constraint booking_member_id_fkey foreign KEY (member_id) references member (id),
  constraint booking_event_id_fkey foreign KEY (event_id) references event (id),
  constraint booking_organization_id_fkey foreign KEY (organization_id) references organization (id)
) TABLESPACE pg_default;

create index IF not exists idx_booking_ticket_class_id on public.booking using btree (ticket_class_id) TABLESPACE pg_default;

create index IF not exists idx_booking_tenant_id on public.booking using btree (tenant_id) TABLESPACE pg_default;
